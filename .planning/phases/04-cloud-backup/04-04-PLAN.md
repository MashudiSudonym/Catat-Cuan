Create an AuthController using Riverpod's @riverpod annotation (per AGENTS.md: initialize in `build()`, NOT constructor). This controller manages sign-in state separately from backup progress.

**Create `lib/presentation/controllers/auth_controller.dart`:**

```dart
import 'package:riverpod_annotation/riverpod_annotation.dart';
import 'package:catat_cuan/domain/entities/backup/auth_user.dart';
import 'package:catat_cuan/presentation/providers/repositories/repository_providers.dart';
import 'package:catat_cuan/presentation/utils/error/error_message_mapper.dart';
import 'package:catat_cuan/data/services/shared_preferences_service.dart';

part 'auth_controller.g.dart';

/// Auth state - either null (not connected) or AuthUser
typedef AuthState = AsyncValue<AuthUser?>;

@riverpod
class AuthController extends _$AuthController {
  @override
  AuthState build() {
    // Check for existing user on init
    _checkExistingUser();
    return const AsyncValue.loading();
  }

  Future<void> _checkExistingUser() async {
    state = const AsyncValue.loading();
    final result = await ref.read(authRepositoryProvider).getCurrentUser();
    if (result.isSuccess) {
      state = AsyncValue.data(result.data);
    } else {
      state = const AsyncValue.data(null);
    }
  }

  Future<void> signIn() async {
    state = const AsyncValue.loading();
    final result = await ref.read(authRepositoryProvider).signIn();
    if (result.isSuccess && result.data != null) {
      // Save connected email to SharedPreferences
      final prefs = SharedPreferencesService();
      await prefs.setConnectedAccountEmail(result.data!.email);
      state = AsyncValue.data(result.data);
    } else if (result.isFailure) {
      state = AsyncValue.error(
        ErrorMessageMapper.getUserMessage(result.failure),
        StackTrace.current,
      );
    } else {
      state = const AsyncValue.data(null);
    }
  }

  Future<void> signOut() async {
    final result = await ref.read(authRepositoryProvider).signOut();
    if (result.isSuccess) {
      final prefs = SharedPreferencesService();
      await prefs.clearBackupMetadata();
      state = const AsyncValue.data(null);
    }
  }
}
```
